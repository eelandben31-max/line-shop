    async function initApp() {
      await liff.init({ liffId: LIFF_ID });

      if (!liff.isLoggedIn()) {
        liff.login();

      }

      const profile = await liff.getProfile();

    function copyMyReferralLink() {
      if (!currentUser) return;
      const shareUrl = `https://liff.line.me/${LIFF_ID}?ref=${currentUser.userId}`;
      navigator.clipboard.writeText(shareUrl).then(() => {

    // 模擬下單並自動計算分潤推播
    async function submitOrder(productName, price) {
      if (!confirm(`確定要預訂【${productName}】嗎？`)) return;

      const orderNo = "ORD" + Date.now();

      // 1. 寫入訂單
      const { data: orderData } = await supabaseClient.from('orders').insert({
        order_number: orderNo,
        buyer_line_id: currentUser.userId,
        total_amount: price,
        status: 'paid'
      }).select().single();

      // 2. 查推薦人並算分潤
      const { data: userData } = await supabaseClient
        .from('users')
        .select('referrer_id')
        .eq('line_user_id', currentUser.userId)
        .single();

      const referrerId = userData?.referrer_id;

      if (referrerId) {
        const commissionAmount = Math.round(price * 0.1); // 10% 分潤

        // 寫入分潤紀錄
        await supabaseClient.from('commissions').insert({
          order_id: orderData.id,
          referrer_id: referrerId,
          commission_amount: commissionAmount,
          is_settled: false
        });

        // 呼叫 LINE 推播給推薦人
        try {
          await fetch('https://api.line.me/v2/bot/message/push', {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'Authorization': `Bearer ${LINE_ACCESS_TOKEN}`
            },
            body: JSON.stringify({
              to: referrerId,
              messages: [{
                type: 'text',
                text: `🎉 好消息！您推薦的顧客【${currentUser.displayName}】已完成預訂【${productName}】！\n獲得分潤：NT$ ${commissionAmount} 元！`
              }]
            })
          });
        } catch (e) {
          console.error("推播發送失敗", e);
        }
      }

      alert("預訂成功！我們會盡快與您聯絡。");
      liff.closeWindow(); // 關閉 LINE 內頁視窗
    }

    initApp();
